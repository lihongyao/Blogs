

1）创建项目

```shell
$ pnpm create vite gen-svg --template react-ts
```

2）集成 tailwindcss

- 安装 Tailwind CSS

  ```shell
  $ pnpm add tailwindcss @tailwindcss/vite
  ```

- 配置 Vite 插件：在 vite.config.ts 配置文件中添加 @tailwindcss/vite 插件

  ```ts
  import { defineConfig } from "vite";
  import react from "@vitejs/plugin-react";
  import tailwindcss from "@tailwindcss/vite";
  // https://vite.dev/config/
  export default defineConfig({
    plugins: [react(), tailwindcss()],
  });
  ```

- 导入 Tailwind CSS：在您的 CSS 文件（*通常是 index.css*）中导入 Tailwind CSS 的内容

  ```css
  @import "tailwindcss";
  ```

3）创建脚本工具，安装相关依赖

```shell
$ pnpm add --save-dev prettier @types/prettier
```

> **`/scripts/generate_svg.ts`**

```ts
import fs from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";
import prettier from "prettier";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// 项目根目录，假设脚本在 gen-svg 目录下
const projectRoot = path.resolve(__dirname, "..");

// SVG 图标目录
const ICONS_DIR = path.join(projectRoot, "public", "icons");

// 输出文件路径
const outputFile = path.join(projectRoot, "src", "components", "svgPath_all.ts");

/**
 * 递归读取指定目录下的所有 SVG 文件
 * @param dir 目录路径
 * @returns SVG 文件完整路径数组
 */
async function walkDir(dir: string): Promise<string[]> {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  const results: string[] = [];

  for (const entry of entries) {
    const fullPath = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      results.push(...(await walkDir(fullPath)));
    } else if (entry.isFile() && fullPath.endsWith(".svg")) {
      results.push(fullPath);
    }
  }

  return results;
}

async function generate() {
  const svgFiles = await walkDir(ICONS_DIR);

  // 转换为相对路径，并统一使用 posix 分隔符 '/'
  const svgNames = svgFiles.map((fullPath) => {
    const relative = path.relative(ICONS_DIR, fullPath);
    const noExt = relative.replace(/\.svg$/i, "");
    return noExt.split(path.sep).join(path.posix.sep);
  });

  const now = new Date();
  const timestamp = now.toISOString(); // ISO 格式时间，方便追踪

  // 生成 TypeScript 文件内容
  const output = `
// ⚠️ 此文件由脚本自动生成，请勿手动修改
// 生成时间: ${timestamp}
export const SVG_PATH_NAMES = [
  ${svgNames.map((n) => `"${n}"`).join(",\n  ")}
] as const;

export type SvgPathName = typeof SVG_PATH_NAMES[number];
`;

  // 使用 Prettier 格式化
  const prettierConfig = await prettier.resolveConfig(projectRoot);
  const formatted = await prettier.format(output, {
    ...prettierConfig,
    parser: "typescript",
  });

  // 确保输出目录存在
  await fs.mkdir(path.dirname(outputFile), { recursive: true });

  // 写入文件
  await fs.writeFile(outputFile, formatted, "utf8");

  console.log(`✔️ 已生成 ${outputFile}`);
}

generate().catch((err) => {
  console.error("❌ 生成 svgPath_all.ts 失败:", err);
  process.exit(1);
});

```

4）设置脚本指令

```json
{
  "gen:svg": "node scripts/generate_svg.ts"
}
```

5）在 **`public/icons`** 目录下放置 `.svg` 文件

6）执行脚本命令 `pnpm gen:svg`，即可生成 svgPath_all.ts 类型声明文件。

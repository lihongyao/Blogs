# 概述

FLV.js 是一个由 Bilibili 开源的 JavaScript 库，用于在现代浏览器中直接播放 FLV（Flash Video）格式的视频，无需依赖 Flash 插件，以下是其核心特性和工作原理：

# 基础概念

**技术原理**

- 通过 Media Source Extensions (MSE) API 将 FLV 流实时转码为 ISO BMFF（MP4 片段），再由浏览器解码播放。
- 支持 HTTP-FLV 和 WebSocket-FLV 协议，适用于直播和点播场景

**FLV 格式**

- FLV 是 Adobe 推出的流媒体容器格式，由 Header 和 Body 组成，常用于在线视频传输

# 核心优势

- **兼容性**：支持 Chrome、Firefox、Safari 10+、IE11 和 Edge 等现代浏览器
- **性能**：利用浏览器硬件加速，支持高清视频和低延迟直播（延迟可低至 1-3 秒）
- **灵活性**：开发者可自定义 UI 和控制逻辑

# 使用示例

```html
<script src="https://cdn.jsdelivr.net/npm/flv.js/dist/flv.min.js"></script>
<script>
  if (flvjs.isSupported()) {
    const player = flvjs.createPlayer({
      type: 'flv',
      url: 'http://example.com/live.flv'
    });
    player.attachMediaElement(document.getElementById('video'));
    player.load();
    player.play();
  }
</script>
```

# 代码

```react
import clsx from 'clsx';
import flvjs from 'flv.js';
import { memo, useEffect, useRef, useState } from 'react';

export default memo(function NVRVideoPreview() {
  // -- 状态管理
  const [videoUrl, setVideoUrl] = useState<string | null>(null);
  const [status, setStatus] = useState('准备加载视频流...');
  const [error, setError] = useState('');
  const [isPlaying, setIsPlaying] = useState(false);

  // -- Refs
  const videoRef = useRef<HTMLVideoElement>(null);
  const flvPlayerRef = useRef<flvjs.Player | null>(null);

  // -- 获取视频流地址
  const handleFetchVideoStream = async () => {
    try {
      setStatus('正在获取视频流地址...');
      setError('');
      const response = await fetch(
        'https://www.bschzhss.com/api/smartsand/nvr/FC8004001',
      );

      if (!response.ok) {
        throw new Error(`HTTP错误! 状态码: ${response.status}`);
      }

      const data = await response.json();

      if (data.code === '0000' && data.data) {
        setVideoUrl(data.data);
        setStatus('视频流地址获取成功!');
        return data.data;
      } else {
        throw new Error('无效的响应数据格式或缺少预览地址');
      }
    } catch (err: any) {
      console.error('获取视频流地址失败:', err);
      setError(`获取视频流地址失败: ${err.message}`);
      setStatus('获取视频流地址失败');
      throw err;
    }
  };

  // -- 初始化FLV播放器
  const initPlayer = (url: string) => {
    if (flvPlayerRef.current) {
      destroyPlayer();
    }

    if (flvjs.isSupported() && videoRef.current) {
      const player = flvjs.createPlayer({
        type: 'flv',
        url: url,
        isLive: true,
      });

      player.attachMediaElement(videoRef.current);
      player.load();

      player.on(flvjs.Events.ERROR, (errorType, errorDetail, errorInfo) => {
        console.error('FLV播放错误:', errorType, errorDetail, errorInfo);
        setError(`播放错误: ${errorType} - ${errorDetail}`);
      });

      player.on(flvjs.Events.METADATA_ARRIVED, () => {
        setStatus('正在播放视频流...');
      });

      flvPlayerRef.current = player;
      return true;
    } else {
      setError('您的浏览器不支持FLV播放');
      return false;
    }
  };

  // -- 销毁播放器
  const destroyPlayer = () => {
    if (flvPlayerRef.current) {
      flvPlayerRef.current.pause();
      flvPlayerRef.current.unload();
      flvPlayerRef.current.detachMediaElement();
      flvPlayerRef.current.destroy();
      flvPlayerRef.current = null;
    }
  };

  // 播放控制
  const handlePlay = () => {
    if (videoUrl) {
      if (initPlayer(videoUrl)) {
        flvPlayerRef.current?.play();
        setIsPlaying(true);
      }
    }
  };

  const handleStop = () => {
    destroyPlayer();
    setIsPlaying(false);
    setStatus('视频流已停止');
  };

  // 组件卸载时清理
  useEffect(() => () => destroyPlayer(), []);

  return (
    <div className="container mx-auto max-w-[1200px] mt-4">
      <h1 className="text-xl">NVR 视频预览</h1>
      <div className="flex gap-4 my-4">
        <button
          className={clsx('py-2 px-4 bg-[#4caf50] text-white rounded-md', {
            'bg-[#ccc] cursor-not-allowed': isPlaying,
          })}
          onClick={handleFetchVideoStream}
        >
          加载视频流
        </button>
        <button
          className={clsx('py-2 px-4 bg-[#4caf50] text-white rounded-md', {
            'bg-[#ccc] cursor-not-allowed': !videoUrl || isPlaying,
          })}
          onClick={handlePlay}
        >
          播放
        </button>
        <button
          className={clsx('py-2 px-4 bg-[#4caf50] text-white rounded-md', {
            'bg-[#ccc] cursor-not-allowed': !isPlaying,
          })}
          onClick={handleStop}
        >
          停止
        </button>
      </div>

      <div className="status">{status}</div>
      {error && <div className="error">{error}</div>}
      <video className="mt-4 aspect-video w-[800px]" ref={videoRef} controls />
    </div>
  );
});

```
